# How to Use Deep Learning for Genomic Sequence Classification in Python

## Question
How can we apply deep learning techniques to classify genomic sequences using Python?

### Step 2: Import libraries

```python
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import LabelEncoder
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense, Embedding, Flatten, LSTM
from Bio import SeqIO
```

### Step 3: Load and Preprocess Data

Assuming your genomic sequences are in FASTA format.

```python
def load_sequences(file_path):
    sequences = []
    labels = []
    for record in SeqIO.parse(file_path, "fasta"):
        sequences.append(str(record.seq))
        labels.append(record.description.split()[1])  # Assuming label is in the description
    return sequences, labels

sequences, labels = load_sequences("genomic_sequences.fasta")

# Encode labels
label_encoder = LabelEncoder()
labels_encoded = label_encoder.fit_transform(labels)

# Train-Test Split
X_train, X_test, y_train, y_test = train_test_split(sequences, labels_encoded, test_size=0.2, random_state=42)
```

### Step 4: Sequence Vectorization

Convert sequences into numerical format.

```python
def create_vocab(sequences):
    vocab = set("".join(sequences))
    return {char: idx + 1 for idx, char in enumerate(vocab)}  # Start indexing from 1

def sequence_to_vector(seq, vocab, max_len=100):
    vector = [vocab[char] for char in seq if char in vocab]
    vector += [0] * (max_len - len(vector))  # Padding
    return vector[:max_len]  # Truncate

vocab = create_vocab(sequences)
X_train_vec = np.array([sequence_to_vector(seq, vocab) for seq in X_train])
X_test_vec = np.array([sequence_to_vector(seq, vocab) for seq in X_test])
```

### Step 5: Build the LSTM Model

```python
model = Sequential()
model.add(Embedding(input_dim=len(vocab) + 1, output_dim=50, input_length=100))  # Adjust embedding size and max length as needed
model.add(LSTM(units=64))
model.add(Dense(len(np.unique(labels_encoded)), activation='softmax'))

model.compile(loss='sparse_categorical_crossentropy', optimizer='adam', metrics=['accuracy'])
```

### Step 6: Train the Model

```python
history = model.fit(X_train_vec, y_train, epochs=20, batch_size=32, validation_data=(X_test_vec, y_test))
```

### Step 7: Evaluate the Model

```python
loss, accuracy = model.evaluate(X_test_vec, y_test)
print(f"Test Accuracy: {accuracy * 100:.2f}%")
```

### Step 8: Make Predictions

```python
def predict_sequence(seq, model, vocab):
    seq_vec = np.array([sequence_to_vector(seq, vocab)])
    pred = model.predict(seq_vec)
    return label_encoder.inverse_transform([np.argmax(pred)])

example_sequence = "ATCGTAGCTAGCT"  # Example sequence
predicted_label = predict_sequence(example_sequence, model, vocab)
print(f"The predicted label for the sequence is: {predicted_label[0]}")
```

### Conclusion

This tutorial provides a fundamental approach to classify genomic sequences using Deep Learning in Python, utilizing an LSTM model.
